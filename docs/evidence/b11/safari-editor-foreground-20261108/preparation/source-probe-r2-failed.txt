// The real pinned source entry point/API parser/probe composed under synthetic JSDOM events.
// This checks harness compatibility, never claims actual Safari execution.
import test from 'node:test';import assert from 'node:assert/strict';import {readFile} from 'node:fs/promises';import {JSDOM,loadModules,until,tick} from './test-support.mjs';import {createFixture,markers} from './fixture.mjs';import {validateObservation} from './observation.mjs';import {privateFlags} from './verify-foreground.mjs';
test('pinned actual main/API/probe compose across held editor foreground revocation',async t=>{
 const f=createFixture(),dom=new JSDOM('<div id="app"></div>',{url:'http://127.0.0.1:12345/view',runScripts:'outside-only',pretendToBeVisual:true});t.after(()=>dom.window.close());const w=dom.window;Object.defineProperty(w.HTMLElement.prototype,'innerText',{get(){return this.textContent;}});let visibility='visible';Object.defineProperty(w.document,'visibilityState',{get:()=>visibility});w.indexedDB={databases:async()=>[]};const observations=[];w.fetch=async(path,opts)=>{assert.equal(path,'/__fixture/observation');observations.push(validateObservation(JSON.parse(opts.body)));return {ok:true};};
 w.eval(await readFile(new URL('./probe.js',import.meta.url),'utf8'));await tick();
 const {API}=loadModules(dom)('api'),api=new API(async(path,opts)=>{const r=await f.handle(path,opts);return {ok:r.status>=200&&r.status<300,status:r.status,json:async()=>r.body};});loadModules(dom,api)('main');const root=w.document.getElementById('app');await until(()=>root.querySelector('#correction')&&root.querySelector('main').getAttribute('aria-busy')==='false');const input=root.querySelector('#correction');input.value=markers.draft;input.dispatchEvent(new w.Event('input',{bubbles:true}));
 const observe=async()=>{const before=observations.length;[...w.document.querySelectorAll('button')].find(b=>b.textContent==='Record fixture observation').click();await until(()=>observations.length===before+1);await tick();return observations.at(-1);};
 const initial=await observe();for(const k of privateFlags)assert.equal(initial.current[k],true,k);
 visibility='hidden';w.document.dispatchEvent(new w.Event('visibilitychange'));f.control({actor:'revoked',hold:true,phase:'held'});visibility='visible';w.document.dispatchEvent(new w.Event('visibilitychange'));await until(()=>f.snapshot().pendingAuthority===1);await new Promise(r=>setTimeout(r,20));const held=await observe();for(const k of privateFlags)assert.equal(held.current[k],false,k);assert(held.current.busy);assert.equal(held.doc,initial.doc);
 f.control({hold:false,phase:'final'});await until(()=>root.querySelector('main').getAttribute('aria-busy')==='false');const final=await observe();for(const k of privateFlags)assert.equal(final.current[k],false,k);assert(final.current.accessEnded&&final.current.refreshAvailable);assert.deepEqual(f.snapshot().counts,{commandAttempts:0,commandEffects:0,forbiddenResponseCount:0});
});
